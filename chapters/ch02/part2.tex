\begin{definition}% 原稿标为「定义 9」
$A=(a_{ij})_{m\times n}$ 复矩阵，共轭
\[
\bar A\overset{\mathrm{def}}{=}(\bar a_{ij})_{m\times n}
\]
\end{definition}

\noindent\textbf{共轭性质}
\begin{align*}
(1)&\quad \overline{\bar A}=A\\
(2)&\quad \overline{A+B}=\bar A+\bar B\\
(3)&\quad \overline{c\cdot A}=\bar c\cdot\bar A\\
(4)&\quad \overline{A\cdot B}=\bar A\cdot\bar B\\
(5)&\quad \overline{A'}=\bar A'% 原稿作 \overline{A}'=\overline{A'}（左端撇号在共轭横线之外），按共轭与转置可交换记
\end{align*}
